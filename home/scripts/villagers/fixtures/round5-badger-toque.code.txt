export default function (villager, api) {
  const { THREE } = api
  const felt = (c) => new THREE.MeshStandardMaterial({ color: c, roughness: 1 })
  const bead = (c) => new THREE.MeshStandardMaterial({ color: c, roughness: 0.25 })
  const add = (parent, geo, mat, x, y, z) => { const m = new THREE.Mesh(geo, mat); m.position.set(x, y, z); parent.add(m); return m }
  const grey = felt('#9a958b'), dark = felt('#4b4643'), cream = felt('#e8dfcf')
  const wood = felt('#b8864e'), crust = felt('#c98f4f'), white = felt('#f4efe4')

  // Stout pear body, wide and low, with a short stubby tail behind.
  add(villager, new THREE.SphereGeometry(0.19, 32, 20), grey, 0, 0.15, 0).scale.set(1.15, 0.8, 1)
  add(villager, new THREE.SphereGeometry(0.05, 16, 12), cream, 0, 0.12, -0.2).scale.set(1, 1, 1.1)
  // Neck, so the head sits forward of the body.
  add(villager, new THREE.CylinderGeometry(0.09, 0.1, 0.1, 18), grey, 0, 0.24, 0.03)

  const head = villager.tag(new THREE.Group(), 'head')
  head.position.set(0, 0.33, 0.05)
  villager.add(head)
  add(head, new THREE.SphereGeometry(0.14, 32, 20), cream, 0, 0, 0).scale.set(1.1, 0.88, 1)
  // Badger face stripes running over the eyes.
  for (const s of [-1, 1]) {
    add(head, new THREE.SphereGeometry(0.142, 24, 16), dark, s * 0.072, 0, 0).scale.set(0.3, 0.9, 1)
  }
  // Short wide muzzle reaching forward, with a glossy nose and a stitched smile.
  add(head, new THREE.SphereGeometry(0.06, 20, 12), cream, 0, -0.035, 0.15).scale.set(1.05, 0.8, 1.6)
  add(head, new THREE.SphereGeometry(0.02, 14, 10), bead('#1a1412'), 0, -0.012, 0.243)
  const smile = add(head, new THREE.TorusGeometry(0.03, 0.004, 6, 16, Math.PI), dark, 0, -0.04, 0.222)
  smile.rotation.z = Math.PI
  smile.rotation.x = -0.5

  // Small round dark ears with cream tips.
  for (const s of [-1, 1]) {
    const ear = add(head, new THREE.SphereGeometry(0.04, 18, 12), dark, s * 0.097, 0.11, -0.012)
    ear.scale.set(1, 1, 0.65)
    add(ear, new THREE.SphereGeometry(0.022, 12, 8), cream, 0, 0.026, 0.004)
  }

  // The cook's mark: a tall puffy toque, a band and a wider puff.
  villager.mark(add(head, new THREE.CylinderGeometry(0.085, 0.08, 0.07, 24), white, 0, 0.125, -0.03))
  villager.mark(add(head, new THREE.SphereGeometry(0.105, 24, 16), white, 0, 0.21, -0.03)).scale.set(1, 0.85, 1)

  for (const s of [-1, 1]) {
    const eye = villager.tag(add(head, new THREE.SphereGeometry(0.02, 14, 10), bead('#14100f'), s * 0.07, 0.025, 0.128), s < 0 ? 'eye-left' : 'eye-right')
    eye.scale.set(1, 1, 0.7)
    const hand = villager.tag(add(villager, new THREE.CapsuleGeometry(0.034, 0.05, 6, 10), dark, s * 0.225, 0.16, 0.05), s < 0 ? 'hand-left' : 'hand-right')
    hand.rotation.z = s * 0.5
    villager.tag(add(villager, new THREE.SphereGeometry(0.05, 16, 10), dark, s * 0.1, 0.03, 0.05), s < 0 ? 'foot-left' : 'foot-right').scale.set(1, 0.65, 1.4)
  }

  // Held prop: a baker's peel with a loaf on the paddle.
  const peel = villager.tag(new THREE.Group(), 'held')
  peel.position.set(-0.23, 0.13, 0.07)
  peel.rotation.z = -0.1
  villager.add(peel)
  add(peel, new THREE.CylinderGeometry(0.011, 0.012, 0.3, 10), wood, 0, 0.08, 0)
  add(peel, new THREE.BoxGeometry(0.13, 0.15, 0.016), wood, 0, 0.27, 0)
  add(peel, new THREE.SphereGeometry(0.05, 18, 12), crust, 0, 0.3, 0.025).scale.set(1.3, 0.8, 0.8)

  villager.gait({ speed: 0.3, stride: 0.2, bob: 0.03, stepped: true, fidget: 0.2 })
  // voice: lower, slower and a touch breathier than the start, for a stout baker who works quietly while the house sleeps.
  villager.about({
    name: 'Brindle', role: 'cook', bark: 'Loaf is proving. Leave the kitchen door shut until six and you will have warm bread.',
    voice: { pitch: 160, range: 2.5, formant: 0.9, rate: 8, breath: 0.15, wobble: 0.2, wobbleRate: 4.5, tilt: 2.1, gain: 0.88 },
  })
}
