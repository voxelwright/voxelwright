export default function (villager, api) {
  const { THREE } = api
  const felt = (c) => new THREE.MeshStandardMaterial({ color: c, roughness: 1 })
  const bead = (c) => new THREE.MeshStandardMaterial({ color: c, roughness: 0.25 })
  const add = (parent, geo, mat, x, y, z) => { const m = new THREE.Mesh(geo, mat); m.position.set(x, y, z); parent.add(m); return m }
  const brown = felt('#8f8372'), quill = felt('#6b5d4e'), cream = felt('#e8dcc4'), wood = felt('#8a6a4a')

  // Low teardrop body, longer than tall
  const body = add(villager, new THREE.SphereGeometry(0.2, 32, 20), brown, 0, 0.18, -0.03)
  body.scale.set(1, 0.85, 1.15)
  add(villager, new THREE.SphereGeometry(0.1, 20, 12), cream, 0, 0.14, 0.12).scale.set(0.9, 0.9, 0.5)

  // Spines: cones across the back and shoulders, sunk into the body
  const rows = [[0.25, 1, 0], [0.75, 6, 2.2], [1.2, 9, 2.5], [1.65, 9, 2.0]]
  for (const [t, n, span] of rows) {
    for (let k = 0; k < n; k++) {
      const az = Math.PI + (n === 1 ? 0 : (k / (n - 1) - 0.5) * 2 * span)
      const dx = Math.sin(t) * Math.sin(az), dy = Math.cos(t), dz = Math.sin(t) * Math.cos(az)
      const geo = new THREE.ConeGeometry(0.03, 0.1, 8)
      geo.rotateX(Math.PI / 2)
      const m = add(villager, geo, quill, dx * 0.19, 0.18 + dy * 0.16, -0.03 + dz * 0.215)
      m.rotation.order = 'YXZ'
      m.rotation.y = Math.atan2(dx, dz)
      m.rotation.x = -Math.asin(dy)
      m.position.x += dx * 0.04; m.position.y += dy * 0.04; m.position.z += dz * 0.04
    }
  }

  // Head with pointed snout
  const head = villager.tag(new THREE.Group(), 'head')
  head.position.set(0, 0.27, 0.15)
  villager.add(head)
  add(head, new THREE.SphereGeometry(0.11, 28, 18), cream, 0, 0, 0).scale.set(1.05, 0.95, 1)
  const snoutGeo = new THREE.ConeGeometry(0.045, 0.12, 14)
  snoutGeo.rotateX(Math.PI / 2)
  add(head, snoutGeo, cream, 0, -0.04, 0.12)
  add(head, new THREE.SphereGeometry(0.02, 12, 8), bead('#2a1e1a'), 0, -0.035, 0.185)
  for (const s of [-1, 1]) {
    add(head, new THREE.SphereGeometry(0.036, 14, 10), brown, s * 0.085, 0.07, -0.015).scale.set(1, 1, 0.6)
    const eye = villager.tag(add(head, new THREE.CylinderGeometry(0.02, 0.02, 0.012, 16), bead('#241a18'), s * 0.05, 0.035, 0.094), s < 0 ? 'eye-left' : 'eye-right')
    eye.rotation.x = Math.PI / 2 - 0.15
    const hand = villager.tag(add(villager, new THREE.CapsuleGeometry(0.034, 0.05, 6, 10), cream, s * 0.22, 0.2, 0.06), s < 0 ? 'hand-left' : 'hand-right')
    hand.rotation.z = s * 0.4
    villager.tag(add(villager, new THREE.SphereGeometry(0.048, 16, 10), cream, s * 0.09, 0.03, 0.05), s < 0 ? 'foot-left' : 'foot-right').scale.set(1, 0.65, 1.3)
  }

  // Feather duster: handle is the held prop, plume is the mark, held out beside the head
  const duster = villager.tag(new THREE.Group(), 'held')
  duster.position.set(0.25, 0, 0.06)
  villager.add(duster)
  add(duster, new THREE.CylinderGeometry(0.011, 0.013, 0.34, 10), wood, 0, 0.29, 0)
  const plume = villager.mark(add(duster, new THREE.SphereGeometry(0.078, 20, 14), felt('#d9c27a'), 0, 0.47, 0))
  plume.scale.set(1, 1.05, 1)

  villager.gait({ speed: 0.5, stride: 0.2, bob: 0.03, stepped: true, fidget: 0.7 })
  // voice: a touch higher and quicker than the start for a busy hummer, with more wobble so the hum wavers.
  villager.about({
    name: 'Burr', role: 'tidier', bark: 'Top shelf is done, the middle one is a matter of time. Who moved the jar?',
    voice: { pitch: 270, range: 3.5, formant: 1.2, rate: 13, breath: 0.1, wobble: 0.6, wobbleRate: 5.5, tilt: 1.5, gain: 0.9 },
  })
}
