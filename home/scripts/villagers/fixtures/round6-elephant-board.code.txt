export default function (villager, api) {
  const { THREE } = api
  const felt = (c) => new THREE.MeshStandardMaterial({ color: c, roughness: 1 })
  const bead = (c) => new THREE.MeshStandardMaterial({ color: c, roughness: 0.25 })
  const add = (parent, geo, mat, x, y, z) => { const m = new THREE.Mesh(geo, mat); m.position.set(x, y, z); parent.add(m); return m }
  const grey = felt('#a9a39c'), pinkish = felt('#cdb7ad'), cream = felt('#efe3c8'), ink = felt('#3a3a3a')

  // Body: heavy pear, with a short tail behind.
  const body = add(villager, new THREE.SphereGeometry(0.16, 28, 18), grey, 0, 0.24, 0)
  body.scale.set(1, 1.1, 1.05)
  add(villager, new THREE.SphereGeometry(0.09, 20, 12), cream, 0, 0.21, 0.1).scale.set(1, 1, 0.4)
  const tail = add(villager, new THREE.CapsuleGeometry(0.011, 0.08, 4, 8), grey, 0, 0.17, -0.17)
  tail.rotation.x = -0.3
  add(villager, new THREE.SphereGeometry(0.02, 10, 8), pinkish, 0, 0.12, -0.19)

  // Head, forward of the body.
  const head = villager.tag(new THREE.Group(), 'head')
  head.position.set(0, 0.38, 0.05)
  villager.add(head)
  add(head, new THREE.SphereGeometry(0.15, 30, 20), grey, 0, 0, 0).scale.set(1, 0.95, 0.95)

  // Big ear flaps flaring out to the sides.
  for (const s of [-1, 1]) {
    const ear = add(head, new THREE.SphereGeometry(0.11, 22, 14), grey, s * 0.17, 0.01, -0.04)
    ear.scale.set(0.25, 1, 0.85)
    ear.rotation.y = s * 0.6
    const inner = add(head, new THREE.SphereGeometry(0.085, 18, 12), pinkish, s * 0.185, 0.0, -0.025)
    inner.scale.set(0.2, 1, 0.8)
    inner.rotation.y = s * 0.6
  }

  // Trunk: one tube bent forward and drooping, narrowing to a blunt tip.
  const tg = new THREE.CylinderGeometry(0.028, 0.055, 0.22, 14, 10)
  const p = tg.attributes.position
  for (let i = 0; i < p.count; i++) {
    const t = (p.getY(i) + 0.11) / 0.22
    p.setXYZ(i, p.getX(i), p.getZ(i) - 0.12 * Math.pow(t, 1.5), 0.2 * t)
  }
  tg.computeVertexNormals()
  add(head, tg, grey, 0, -0.04, 0.1)

  // Short tusks either side of the trunk root.
  for (const s of [-1, 1]) {
    const tusk = add(head, new THREE.ConeGeometry(0.022, 0.12, 10), cream, s * 0.08, -0.07, 0.13)
    tusk.rotation.x = Math.PI / 2 + 0.5
  }

  // Eyes.
  for (const s of [-1, 1]) {
    villager.tag(add(head, new THREE.SphereGeometry(0.02, 14, 10), bead('#241a18'), s * 0.07, 0.025, 0.115), s < 0 ? 'eye-left' : 'eye-right')
  }

  // Tutor's mark: skullcap, flat square board lying on it, tassel at a back corner.
  villager.mark(add(head, new THREE.SphereGeometry(0.14, 24, 12), ink, 0, 0.1, 0)).scale.set(1, 0.45, 1)
  villager.mark(add(head, new api.RoundedBoxGeometry(0.44, 0.02, 0.44, 2, 0.005), ink, 0, 0.16, -0.01))
  villager.mark(add(head, new THREE.CylinderGeometry(0.008, 0.008, 0.09, 8), ink, 0.2, 0.115, -0.2))
  villager.mark(add(head, new THREE.SphereGeometry(0.022, 12, 10), ink, 0.2, 0.065, -0.2)).scale.set(1, 1.4, 1)

  // Hands and feet.
  for (const s of [-1, 1]) {
    const hand = villager.tag(add(villager, new THREE.SphereGeometry(0.036, 14, 10), grey, s * 0.17, 0.24, 0.03), s < 0 ? 'hand-left' : 'hand-right')
    hand.scale.set(1, 1.1, 1)
    villager.tag(add(villager, new THREE.CylinderGeometry(0.052, 0.058, 0.09, 14), grey, s * 0.085, 0.045, 0.02), s < 0 ? 'foot-left' : 'foot-right')
  }

  // Held prop: a rolled history scroll with a ribbon, in the left hand.
  const scroll = villager.tag(new THREE.Group(), 'held')
  scroll.position.set(-0.19, 0.24, 0.05)
  scroll.rotation.x = 1.0
  villager.add(scroll)
  add(scroll, new THREE.CylinderGeometry(0.03, 0.03, 0.24, 16), felt('#e8d7b0'), 0, 0, 0)
  add(scroll, new THREE.CylinderGeometry(0.034, 0.034, 0.02, 16), felt('#8a6a4a'), 0, 0.12, 0)
  add(scroll, new THREE.CylinderGeometry(0.034, 0.034, 0.02, 16), felt('#8a6a4a'), 0, -0.12, 0)
  add(scroll, new THREE.CylinderGeometry(0.033, 0.033, 0.025, 16), felt('#8a6a4a'), 0, 0.02, 0)

  villager.gait({ speed: 0.26, stride: 0.18, bob: 0.02, stepped: true, fidget: 0.2 })
  // voice: lower, slower and darker than the start for a large, old, unhurried teacher; a slower wobble for age.
  villager.about({
    name: 'Ambrose', role: 'tutor', bark: 'Ask me why before you ask me when, and the dates will stay put.',
    voice: { pitch: 135, range: 2.5, formant: 0.85, rate: 6.5, breath: 0.25, wobble: 0.7, wobbleRate: 4.5, tilt: 2.2, gain: 1.0 },
  })
}
