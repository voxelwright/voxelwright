export default function (villager, api) {
  const { THREE } = api
  const felt = (c) => new THREE.MeshStandardMaterial({ color: c, roughness: 1 })
  const bead = (c) => new THREE.MeshStandardMaterial({ color: c, roughness: 0.25 })
  const add = (parent, geo, mat, x, y, z) => { const m = new THREE.Mesh(geo, mat); m.position.set(x, y, z); parent.add(m); return m }

  const grey = felt('#b9c0bc'), slate = felt('#7f8f8c'), darkSlate = felt('#5f6d6b')
  const cream = felt('#efe8d6'), legCol = felt('#8a7f6a'), beakCol = felt('#b8693f')
  const board = felt('#3f4650'), wood = felt('#a8703a'), green = felt('#4f6b5e')
  const accent = '#e0b53a'

  // Body: slim upright egg, standing high on long legs.
  add(villager, new THREE.SphereGeometry(0.1, 28, 18), grey, 0, 0.4, 0).scale.set(0.9, 1.4, 0.85)
  add(villager, new THREE.SphereGeometry(0.07, 20, 12), cream, 0, 0.4, 0.07).scale.set(0.8, 1.1, 0.35)
  // Tail feathers sweeping back and down.
  const tail = add(villager, new THREE.ConeGeometry(0.04, 0.14, 12), slate, 0, 0.31, -0.12)
  tail.rotation.x = -Math.PI / 2 - 0.4
  tail.scale.set(1.6, 1, 0.5)
  // Folded wings at the sides.
  for (const s of [-1, 1]) {
    const wing = add(villager, new THREE.CapsuleGeometry(0.04, 0.14, 6, 12), slate, s * 0.095, 0.4, -0.01)
    wing.scale.set(0.55, 1, 1.5)
    wing.rotation.z = s * 0.06
  }
  // Long neck, leaning very slightly forward.
  const neck = add(villager, new THREE.CapsuleGeometry(0.032, 0.2, 6, 12), cream, 0, 0.62, 0)
  neck.rotation.x = 0.1

  // Scarf in the owner's colour, with two tails in front.
  const scarf = add(villager, new THREE.TorusGeometry(0.055, 0.03, 10, 28), felt(accent), 0, 0.57, 0)
  scarf.rotation.x = Math.PI / 2
  for (const s of [-1, 1]) {
    const t = add(villager, new THREE.CapsuleGeometry(0.026, s < 0 ? 0.07 : 0.05, 6, 10), felt(accent), s * 0.03, s < 0 ? 0.51 : 0.52, 0.075)
    t.rotation.z = s * 0.1
  }

  // Head with a long dagger beak.
  const head = villager.tag(new THREE.Group(), 'head')
  head.position.set(0, 0.76, 0.02)
  villager.add(head)
  add(head, new THREE.SphereGeometry(0.06, 24, 16), cream, 0, 0, 0).scale.set(0.95, 1, 1.1)
  const beak = add(head, new THREE.ConeGeometry(0.026, 0.2, 14), beakCol, 0, -0.005, 0.15)
  beak.rotation.x = Math.PI / 2
  for (const s of [-1, 1]) {
    const eye = villager.tag(add(head, new THREE.CylinderGeometry(0.017, 0.017, 0.012, 16), bead('#241a18'), s * 0.042, 0.012, 0.044), s < 0 ? 'eye-left' : 'eye-right')
    eye.rotation.x = Math.PI / 2
  }
  // The tutor's mark: a flat mortarboard, level, with a tassel off one corner.
  add(head, new THREE.CylinderGeometry(0.058, 0.06, 0.03, 20), board, 0, 0.055, 0)
  add(head, new THREE.BoxGeometry(0.155, 0.012, 0.155), board, 0, 0.078, 0)
  add(head, new THREE.CylinderGeometry(0.004, 0.004, 0.05, 6), felt(accent), 0.075, 0.048, 0.073)
  add(head, new THREE.CapsuleGeometry(0.014, 0.03, 6, 8), felt(accent), 0.075, 0.012, 0.073)

  // Hands and long legs with splayed toes.
  for (const s of [-1, 1]) {
    const hand = villager.tag(add(villager, new THREE.CapsuleGeometry(0.026, 0.04, 6, 10), darkSlate, s * 0.13, 0.37, 0.04), s < 0 ? 'hand-left' : 'hand-right')
    hand.rotation.z = s * 0.2
    const foot = villager.tag(new THREE.Group(), s < 0 ? 'foot-left' : 'foot-right')
    foot.position.set(s * 0.06, 0, 0)
    villager.add(foot)
    add(foot, new THREE.CylinderGeometry(0.012, 0.014, 0.31, 8), legCol, 0, 0.155, 0)
    for (const a of [-0.45, 0, 0.45]) {
      const g = new THREE.Group()
      g.rotation.y = a
      foot.add(g)
      const toe = add(g, new THREE.CapsuleGeometry(0.01, 0.06, 4, 8), legCol, 0, 0.01, 0.04)
      toe.rotation.x = Math.PI / 2
    }
  }

  // Prop: a chalk slate with a sum on it.
  const slateProp = villager.tag(new THREE.Group(), 'held')
  slateProp.position.set(-0.16, 0.36, 0.08)
  slateProp.rotation.z = 0.08
  villager.add(slateProp)
  add(slateProp, new api.RoundedBoxGeometry(0.22, 0.3, 0.025, 3, 0.008), wood, 0, 0, 0)
  add(slateProp, new THREE.BoxGeometry(0.19, 0.27, 0.012), green, 0, 0, 0.008)
  const chalk = felt('#f4efe4')
  add(slateProp, new THREE.BoxGeometry(0.05, 0.008, 0.004), chalk, -0.04, 0.07, 0.015)
  add(slateProp, new THREE.BoxGeometry(0.008, 0.05, 0.004), chalk, -0.04, 0.07, 0.015)
  add(slateProp, new THREE.BoxGeometry(0.06, 0.008, 0.004), chalk, 0.03, -0.015, 0.015)
  add(slateProp, new THREE.BoxGeometry(0.06, 0.008, 0.004), chalk, 0.03, -0.045, 0.015)

  villager.gait({ speed: 0.3, stride: 0.3, bob: 0.02, stepped: true, fidget: 0.2 })
  // voice: a little lower and slower than the start, with a flatter range and rounder tone for a measured, strict tutor.
  villager.about({
    name: 'Sedge', role: 'tutor', bark: 'Homework is at four. Bring the fractions and leave the excuses at the door.',
    voice: { pitch: 175, range: 1.2, formant: 0.95, rate: 9, breath: 0.06, wobble: 0.1, wobbleRate: 5, tilt: 1.3, gain: 0.9 },
    persona: {
      role: 'Tutors the children in maths',
      speciality: 'Turning a stuck sum into three small steps',
      temperament: 'dry', register: 'formal', humour: 'dry',
      quirks: ['Asks for the working before accepting any answer', 'Stands very still on one leg while a child thinks', 'Rounds every estimate aloud to the nearest ten'],
      topics: ['times tables', 'fractions and ratios', 'the homework schedule', 'mental arithmetic', 'checking your working'],
      dials: { warmth: 0.45, playful: 0.2, chatty: 0.35, humour: 'clean' },
    },
  })
}
