export default function (villager, api) {
  const { THREE } = api
  const felt = (c) => new THREE.MeshStandardMaterial({ color: c, roughness: 1 })
  const bead = (c) => new THREE.MeshStandardMaterial({ color: c, roughness: 0.25 })
  const add = (parent, geo, mat, x, y, z) => { const m = new THREE.Mesh(geo, mat); m.position.set(x, y, z); parent.add(m); return m }
  const grey = felt('#c9bcab'), pink = felt('#e3a9a3'), cream = felt('#f3e7d1')
  const rust = felt('#b5654a'), brown = felt('#8a6445')
  const ACC = '#3d7cc8'

  // Body
  add(villager, new THREE.SphereGeometry(0.17, 32, 20), grey, 0, 0.15, 0).scale.set(1, 0.8, 0.9)
  add(villager, new THREE.SphereGeometry(0.08, 20, 12), cream, 0, 0.13, 0.11).scale.set(0.9, 0.9, 0.4)
  // Tail
  const tail = add(villager, new THREE.CapsuleGeometry(0.013, 0.2, 6, 8), pink, 0, 0.07, -0.22)
  tail.rotation.x = Math.PI / 2

  // Scarf in the owner's colour
  const scarf = add(villager, new THREE.TorusGeometry(0.12, 0.03, 10, 28), felt(ACC), 0, 0.255, 0)
  scarf.rotation.x = Math.PI / 2
  for (const s of [-1, 1]) add(villager, new THREE.CapsuleGeometry(0.027, 0.07, 6, 10), felt(ACC), s * 0.04, 0.19, 0.13).rotation.z = s * 0.12

  // Mail bag on the +x hip, with strap across the body
  const strap = add(villager, new THREE.BoxGeometry(0.026, 0.31, 0.02), brown, 0.06, 0.18, 0.14)
  strap.rotation.z = 1.107
  add(villager, new api.RoundedBoxGeometry(0.1, 0.12, 0.13, 3, 0.02), rust, 0.225, 0.09, 0)
  const flap = add(villager, new THREE.CylinderGeometry(0.055, 0.055, 0.012, 24), felt(ACC), 0.225, 0.105, 0.068)
  flap.rotation.x = Math.PI / 2
  add(villager, new THREE.SphereGeometry(0.014, 12, 8), bead('#e8d9b0'), 0.225, 0.085, 0.078)

  // Head
  const head = villager.tag(new THREE.Group(), 'head')
  head.position.set(0, 0.36, 0)
  villager.add(head)
  add(head, new THREE.SphereGeometry(0.14, 32, 20), grey, 0, 0, 0).scale.set(1.0, 0.95, 0.95)
  add(head, new THREE.SphereGeometry(0.06, 20, 12), cream, 0, -0.055, 0.1).scale.set(1, 0.8, 0.9)
  add(head, new THREE.SphereGeometry(0.022, 14, 10), bead('#5a3030'), 0, -0.04, 0.15)
  const smile = add(head, new THREE.TorusGeometry(0.02, 0.004, 8, 16, Math.PI), felt('#5a3a34'), 0, -0.065, 0.143)
  smile.rotation.z = Math.PI
  // Peaked cap: low crown, short brim above the eyes
  add(head, new THREE.SphereGeometry(0.13, 24, 12, 0, Math.PI * 2, 0, Math.PI / 2), rust, 0, 0.065, -0.005).scale.set(1, 0.8, 1)
  const visor = add(head, new THREE.CylinderGeometry(0.07, 0.07, 0.012, 20), rust, 0, 0.075, 0.14)
  visor.scale.set(1, 1, 0.8)
  visor.rotation.x = -0.1
  // Ears
  for (const s of [-1, 1]) {
    const ear = add(head, new THREE.SphereGeometry(0.07, 20, 12), grey, s * 0.115, 0.075, -0.03)
    ear.scale.set(1, 1, 0.35)
    add(ear, new THREE.SphereGeometry(0.048, 16, 10), pink, 0, 0, 0.02).scale.set(1, 1, 0.4)
    const eye = villager.tag(add(head, new THREE.CylinderGeometry(0.022, 0.022, 0.012, 18), bead('#241a18'), s * 0.06, 0, 0.12), s < 0 ? 'eye-left' : 'eye-right')
    eye.rotation.x = Math.PI / 2
    const hand = villager.tag(add(villager, new THREE.CapsuleGeometry(0.032, 0.05, 6, 10), grey, s * 0.185, 0.15, 0.03), s < 0 ? 'hand-left' : 'hand-right')
    hand.rotation.z = s * 0.5
    villager.tag(add(villager, new THREE.SphereGeometry(0.045, 16, 10), pink, s * 0.09, 0.03, 0.04), s < 0 ? 'foot-left' : 'foot-right').scale.set(1, 0.65, 1.3)
  }

  // Held letter in the -x hand
  const letter = villager.tag(new THREE.Group(), 'held')
  letter.position.set(-0.27, 0.17, 0.085)
  letter.rotation.z = 0.2
  villager.add(letter)
  add(letter, new THREE.BoxGeometry(0.14, 0.2, 0.012), cream, 0, 0, 0)
  add(letter, new THREE.BoxGeometry(0.1, 0.012, 0.004), felt('#a8956f'), 0, 0.03, 0.008)
  add(letter, new THREE.BoxGeometry(0.1, 0.012, 0.004), felt('#a8956f'), 0, 0.0, 0.008)
  add(letter, new THREE.CylinderGeometry(0.016, 0.016, 0.006, 14), rust, 0, -0.06, 0.009).rotation.x = Math.PI / 2

  villager.gait({ speed: 0.55, stride: 0.2, bob: 0.05, stepped: true, fidget: 1 })
  // voice: kept the high, bright start; more wobble and a faster wobble rate for a jittery mouse, slightly less round tilt.
  villager.about({
    name: 'Pip Postal', role: 'messenger', bark: 'Two letters, one parcel, and a bill with your name on it. Bill first or last?',
    voice: { pitch: 453, range: 5, formant: 1.58, rate: 16, breath: 0.1, wobble: 0.8, wobbleRate: 8, tilt: 1.4, gain: 0.9 },
    persona: {
      role: 'Sorts and delivers the household post', speciality: 'Sorting mail and chasing replies fast',
      temperament: 'upbeat', register: 'casual', humour: 'wordplay',
      quirks: ['Sorts everything by urgency, then by envelope colour', 'Narrates each delivery as it happens', 'Counts the stamps twice'],
      topics: ['what came in the post', 'overdue replies', 'parcel tracking', 'who wrote to whom', 'bills due this week'],
      dials: { warmth: 0.6, playful: 0.6, chatty: 1 },
    },
  })
}
