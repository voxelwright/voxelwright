export default function (villager, api) {
  const { THREE } = api
  const felt = (c) => new THREE.MeshStandardMaterial({ color: c, roughness: 1 })
  const bead = (c) => new THREE.MeshStandardMaterial({ color: c, roughness: 0.25 })
  const add = (parent, geo, mat, x, y, z) => { const m = new THREE.Mesh(geo, mat); m.position.set(x, y, z); parent.add(m); return m }
  const hide = felt('#a9917c'), cream = felt('#ecdfc6'), muzzleC = felt('#e3c9b6'), pink = felt('#d6a29c')
  const hoofC = felt('#5c4b42'), wood = felt('#b58a55'), steel = felt('#8d9096')
  const dark = bead('#2a1e1a')

  // Body: broad barrel with a shoulder hump behind the head
  add(villager, new THREE.SphereGeometry(0.2, 32, 20), hide, 0, 0.24, 0).scale.set(1.15, 0.95, 0.95)
  add(villager, new THREE.SphereGeometry(0.14, 24, 16), hide, 0, 0.34, -0.06).scale.set(1.2, 0.95, 1)
  add(villager, new THREE.SphereGeometry(0.1, 20, 12), muzzleC, 0, 0.2, 0.15).scale.set(1.3, 0.9, 0.4)
  // Thick neck reaching forward to the head
  add(villager, new THREE.CylinderGeometry(0.1, 0.115, 0.14, 20), hide, 0, 0.34, 0.1)

  // Head, forward of the body
  const head = villager.tag(new THREE.Group(), 'head')
  head.position.set(0, 0.38, 0.17)
  villager.add(head)
  add(head, new THREE.SphereGeometry(0.14, 28, 18), hide, 0, 0, 0).scale.set(1.1, 0.95, 0.95)
  add(head, new THREE.SphereGeometry(0.04, 16, 10), cream, 0, 0.115, 0.02).scale.set(1.4, 0.8, 1)
  // Broad muzzle sticking out in front
  add(head, new THREE.RoundedBoxGeometry(0.17, 0.09, 0.1, 4, 0.035), muzzleC, 0, -0.045, 0.13)
  for (const s of [-1, 1]) add(head, new THREE.SphereGeometry(0.0125, 10, 8), dark, s * 0.036, -0.03, 0.182)
  const smile = add(head, new THREE.TorusGeometry(0.032, 0.005, 6, 14, Math.PI), hoofC, 0, -0.058, 0.18)
  smile.rotation.z = Math.PI

  for (const s of [-1, 1]) {
    // Droopy side ears
    const ear = add(head, new THREE.CapsuleGeometry(0.032, 0.09, 6, 10), hide, s * 0.2, -0.005, -0.02)
    ear.rotation.z = -s * 1.97
    ear.scale.set(1, 1, 0.5)
    add(ear, new THREE.CapsuleGeometry(0.018, 0.06, 6, 8), pink, 0, 0, 0.012).scale.set(1, 1, 0.5)
    // Horns: out, then up
    const horn = add(head, new THREE.CapsuleGeometry(0.03, 0.05, 6, 10), cream, s * 0.19, 0.1, -0.03)
    horn.rotation.z = -s * 1.3
    const tip = add(head, new THREE.ConeGeometry(0.03, 0.08, 14), cream, s * 0.275, 0.14, -0.03)
    tip.rotation.z = -s * 0.2
    // Eyes
    const eye = villager.tag(add(head, new THREE.CylinderGeometry(0.02, 0.02, 0.012, 18), dark, s * 0.065, 0.03, 0.118), s < 0 ? 'eye-left' : 'eye-right')
    eye.rotation.x = Math.PI / 2 - 0.15
    // Hands
    const hand = villager.tag(add(villager, new THREE.CapsuleGeometry(0.04, 0.05, 6, 10), hide, s * 0.25, 0.25, 0.06), s < 0 ? 'hand-left' : 'hand-right')
    hand.rotation.z = s * 0.4
    // Solid legs with hooves, belonging to the feet
    const foot = villager.tag(new THREE.Group(), s < 0 ? 'foot-left' : 'foot-right')
    foot.position.set(s * 0.11, 0, 0.02)
    villager.add(foot)
    add(foot, new THREE.CylinderGeometry(0.058, 0.064, 0.11, 16), hide, 0, 0.085, 0)
    add(foot, new THREE.SphereGeometry(0.055, 16, 10), hoofC, 0, 0.033, 0.015).scale.set(1.05, 0.6, 1.25)
  }

  // The maker's mark: headband, hinge, short arm and an upright magnifying ring beside the head
  const band = add(head, new THREE.TorusGeometry(0.137, 0.013, 8, 32), cream, 0, 0.07, 0)
  band.rotation.x = Math.PI / 2
  band.scale.set(1, 0.9, 1)
  villager.mark(band)
  villager.mark(add(head, new THREE.SphereGeometry(0.025, 12, 8), cream, 0.14, 0.07, 0.03))
  villager.mark(add(head, new THREE.CapsuleGeometry(0.016, 0.03, 6, 8), cream, 0.185, 0.09, 0.05))
  villager.mark(add(head, new THREE.TorusGeometry(0.07, 0.022, 12, 32), cream, 0.2, 0.19, 0.06))

  // Prop: a wooden mallet, held in the left mitten
  const mallet = villager.tag(new THREE.Group(), 'held')
  mallet.position.set(-0.25, 0.21, 0.06)
  mallet.rotation.z = 0.1
  villager.add(mallet)
  add(mallet, new THREE.CylinderGeometry(0.014, 0.017, 0.3, 10), wood, 0, 0.07, 0)
  add(mallet, new THREE.RoundedBoxGeometry(0.13, 0.075, 0.075, 3, 0.02), wood, 0, 0.21, 0)
  for (const s of [-1, 1]) add(mallet, new THREE.CylinderGeometry(0.04, 0.04, 0.012, 14), steel, s * 0.066, 0.21, 0).rotation.z = Math.PI / 2

  villager.gait({ speed: 0.28, stride: 0.2, bob: 0.03, stepped: true, fidget: 0.3 })
  // voice: kept low and dark for a burly ox, but quicker (rate 11) and a bit more range (3.6) for the mouthy sofa-buddy patter.
  villager.about({
    name: 'Brindle', role: 'maker', bark: 'That chair has been lying to you for months. Give me ten minutes and some glue.',
    voice: { pitch: 134.7, range: 3.6, formant: 0.79, rate: 11, breath: 0.1, wobble: 0.25, wobbleRate: 5, tilt: 2.2, gain: 0.9 },
  })
}
