export default function (villager, api) {
  const { THREE } = api
  const felt = (c) => new THREE.MeshStandardMaterial({ color: c, roughness: 1 })
  const bead = (c) => new THREE.MeshStandardMaterial({ color: c, roughness: 0.25 })
  const add = (parent, geo, mat, x, y, z) => { const m = new THREE.Mesh(geo, mat); m.position.set(x, y, z); parent.add(m); return m }
  const coat = felt('#e0c79c'), spot = felt('#a98456'), cream = felt('#f1e4c8'), dark = felt('#6b5340')

  // Body: pear shape, set on legs.
  const body = add(villager, new THREE.SphereGeometry(0.15, 28, 18), coat, 0, 0.26, -0.01)
  body.scale.set(0.95, 0.85, 1.15)
  add(villager, new THREE.SphereGeometry(0.04, 12, 8), spot, 0.09, 0.32, -0.08).scale.set(1, 1, 0.5)
  add(villager, new THREE.SphereGeometry(0.045, 12, 8), spot, -0.1, 0.25, -0.02).scale.set(0.5, 1, 1)
  add(villager, new THREE.SphereGeometry(0.04, 12, 8), spot, 0.05, 0.22, -0.15).scale.set(1, 1, 0.5)
  add(villager, new THREE.SphereGeometry(0.035, 12, 8), spot, -0.05, 0.34, -0.1).scale.set(1, 1, 0.5)

  // Long neck, leaning forward a little.
  const neck = add(villager, new THREE.CylinderGeometry(0.045, 0.06, 0.16, 18), coat, 0, 0.37, 0.015)
  neck.rotation.x = 0.2
  add(villager, new THREE.SphereGeometry(0.03, 12, 8), spot, 0.035, 0.37, 0.02).scale.set(0.5, 1, 1)
  add(villager, new THREE.SphereGeometry(0.03, 12, 8), spot, -0.035, 0.34, 0.01).scale.set(0.5, 1, 1)
  for (const [y, z] of [[0.33, -0.04], [0.39, -0.02], [0.44, 0.0]]) add(villager, new THREE.SphereGeometry(0.02, 10, 8), dark, 0, y, z)

  // Tail with a tuft.
  const tail = add(villager, new THREE.CylinderGeometry(0.008, 0.008, 0.14, 8), coat, 0, 0.26, -0.2)
  tail.rotation.x = -0.5
  add(villager, new THREE.SphereGeometry(0.02, 10, 8), dark, 0, 0.2, -0.25)

  // Head.
  const head = villager.tag(new THREE.Group(), 'head')
  head.position.set(0, 0.47, 0.035)
  villager.add(head)
  add(head, new THREE.SphereGeometry(0.065, 24, 16), coat, 0, 0, 0).scale.set(1, 0.95, 1.05)
  add(head, new THREE.SphereGeometry(0.045, 18, 12), cream, 0, -0.022, 0.07).scale.set(1, 0.8, 1.3)
  add(head, new THREE.SphereGeometry(0.011, 10, 8), bead('#3a2a22'), 0, -0.008, 0.12)
  for (const s of [-1, 1]) {
    const oss = add(head, new THREE.CylinderGeometry(0.014, 0.018, 0.07, 10), coat, s * 0.036, 0.075, -0.015)
    oss.rotation.z = -s * 0.12
    add(head, new THREE.SphereGeometry(0.02, 12, 8), dark, s * 0.043, 0.112, -0.015)
    const ear = add(head, new THREE.CapsuleGeometry(0.018, 0.05, 4, 8), coat, s * 0.085, 0.02, -0.02)
    ear.rotation.z = s * 1.2
    ear.scale.set(1, 1, 0.5)
    const eye = villager.tag(add(head, new THREE.CylinderGeometry(0.014, 0.014, 0.01, 14), bead('#241a18'), s * 0.038, 0.022, 0.055), s < 0 ? 'eye-left' : 'eye-right')
    eye.rotation.x = Math.PI / 2 - 0.2
    // Legs belong to the feet.
    const foot = villager.tag(new THREE.Group(), s < 0 ? 'foot-left' : 'foot-right')
    foot.position.set(s * 0.07, 0, 0.03)
    villager.add(foot)
    add(foot, new THREE.CylinderGeometry(0.026, 0.03, 0.16, 12), coat, 0, 0.09, 0)
    add(foot, new THREE.SphereGeometry(0.036, 12, 8), dark, 0, 0.02, 0.008).scale.set(1, 0.65, 1.3)
    // Hands.
    const hand = villager.tag(add(villager, new THREE.CapsuleGeometry(0.03, 0.04, 6, 10), coat, s * 0.17, 0.24, 0.05), s < 0 ? 'hand-left' : 'hand-right')
    hand.rotation.z = s * 0.4
  }

  // Held feather duster: handle in the right hand, round plume high above.
  const duster = villager.tag(new THREE.Group(), 'held')
  duster.position.set(0.2, 0.2, 0.05)
  villager.add(duster)
  add(duster, new THREE.CylinderGeometry(0.011, 0.013, 0.34, 8), felt('#a8703a'), 0, 0.17, 0)
  const plume = add(duster, new THREE.SphereGeometry(0.095, 22, 16), cream, 0, 0.425, 0)
  plume.scale.set(1, 1, 0.9)
  villager.mark(plume)

  villager.gait({ speed: 0.3, stride: 0.3, bob: 0.03, stepped: true, fidget: 0.6 })
  // voice: slightly lower formant for a long throat, slower and more deliberate rate, a touch rounder and steadier for a fussy, precise speaker.
  villager.about({
    name: 'Stilt', role: 'tidier', bark: 'Top shelf, left. The gravy boat has been facing the wrong way since March.',
    voice: { pitch: 205, range: 2.6, formant: 0.92, rate: 8, breath: 0.1, wobble: 0.2, wobbleRate: 5, tilt: 1.8, gain: 0.9 },
  })
}
