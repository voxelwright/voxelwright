export default function (villager, api) {
  const { THREE, RoundedBoxGeometry } = api
  const felt = (c) => new THREE.MeshStandardMaterial({ color: c, roughness: 1 })
  const bead = (c) => new THREE.MeshStandardMaterial({ color: c, roughness: 0.25 })
  const add = (parent, geo, mat, x, y, z) => { const m = new THREE.Mesh(geo, mat); m.position.set(x, y, z); parent.add(m); return m }
  const honey = felt('#b98a5e'), cream = felt('#f2e2c6'), mustard = felt('#d9a84a'), dark = felt('#5a4636')
  const iron = felt('#6b625c'), pancake = felt('#e6bd7e')

  // big pear-shaped body
  add(villager, new THREE.SphereGeometry(0.21, 32, 20), honey, 0, 0.24, 0).scale.set(1.05, 1.1, 0.92)
  // cream belly and apron
  add(villager, new THREE.SphereGeometry(0.11, 20, 12), cream, 0, 0.2, 0.15).scale.set(1, 1.1, 0.35)
  const apron = add(villager, new RoundedBoxGeometry(0.17, 0.15, 0.03, 3, 0.012), mustard, 0, 0.17, 0.19)
  apron.rotation.x = -0.1
  add(villager, new THREE.SphereGeometry(0.03, 12, 8), mustard, 0, 0.215, 0.2).scale.set(1.6, 0.9, 0.5)
  // round tail
  add(villager, new THREE.SphereGeometry(0.05, 16, 10), cream, 0, 0.1, -0.19)

  // scarf (accent only)
  const scarf = add(villager, new THREE.TorusGeometry(0.15, 0.03, 10, 28), felt('#c8553d'), 0, 0.45, 0)
  scarf.rotation.x = Math.PI / 2
  const tail = add(villager, new THREE.CapsuleGeometry(0.025, 0.07, 6, 10), felt('#c8553d'), 0.08, 0.4, 0.13)
  tail.rotation.z = 0.2

  // head
  const head = villager.tag(new THREE.Group(), 'head')
  head.position.set(0, 0.41, 0)
  villager.add(head)
  add(head, new THREE.SphereGeometry(0.15, 32, 20), honey, 0, 0, 0).scale.set(1.1, 0.95, 0.98)
  add(head, new THREE.SphereGeometry(0.065, 20, 12), cream, 0, -0.04, 0.11).scale.set(1.2, 0.85, 0.7)
  add(head, new THREE.SphereGeometry(0.02, 12, 8), bead('#2e211b'), 0, -0.012, 0.165).scale.set(1.3, 0.9, 0.9)
  // stitched smile
  const smile = add(head, new THREE.TorusGeometry(0.025, 0.004, 6, 14, Math.PI), dark, 0, -0.045, 0.16)
  smile.rotation.z = Math.PI
  // round ears
  for (const s of [-1, 1]) {
    add(head, new THREE.SphereGeometry(0.05, 18, 12), honey, s * 0.115, 0.1, -0.01).scale.set(1, 1, 0.6)
    add(head, new THREE.SphereGeometry(0.03, 14, 10), cream, s * 0.115, 0.1, 0.012).scale.set(1, 1, 0.5)
    const eye = villager.tag(add(head, new THREE.CylinderGeometry(0.019, 0.019, 0.012, 18), bead('#241a18'), s * 0.06, 0.03, 0.135), s < 0 ? 'eye-left' : 'eye-right')
    eye.rotation.x = Math.PI / 2 - 0.15
  }
  // chef toque
  add(head, new THREE.CylinderGeometry(0.095, 0.1, 0.06, 24), cream, 0, 0.145, 0)
  add(head, new THREE.SphereGeometry(0.075, 22, 14), cream, 0, 0.195, 0).scale.set(1.15, 0.9, 1.15)

  // hands and feet
  for (const s of [-1, 1]) {
    const hand = villager.tag(add(villager, new THREE.CapsuleGeometry(0.036, 0.05, 6, 10), honey, s * 0.225, 0.25, 0.03), s < 0 ? 'hand-left' : 'hand-right')
    hand.rotation.z = s * 0.5
    villager.tag(add(villager, new THREE.SphereGeometry(0.058, 16, 10), honey, s * 0.115, 0.03, 0.03), s < 0 ? 'foot-left' : 'foot-right').scale.set(1, 0.65, 1.3)
  }

  // frying pan with pancake
  const pan = villager.tag(new THREE.Group(), 'held')
  pan.position.set(-0.235, 0.22, 0.09)
  villager.add(pan)
  add(pan, new THREE.CylinderGeometry(0.07, 0.062, 0.025, 24), iron, 0, 0, 0.07)
  add(pan, new THREE.CylinderGeometry(0.055, 0.055, 0.02, 22), pancake, 0, 0.015, 0.07)
  const handle = add(pan, new THREE.CylinderGeometry(0.01, 0.012, 0.12, 8), felt('#a8703a'), 0, 0, -0.03)
  handle.rotation.x = Math.PI / 2

  villager.gait({ speed: 0.26, stride: 0.18, bob: 0.03, stepped: true, fidget: 0.4 })
  villager.about({
    name: 'Biscuit',
    role: 'cook',
    bark: "Sunday breakfast is almost ready. Pancakes, or eggs first?",
    voice: { pitch: 130, range: 2.5, formant: 0.82, rate: 6, breath: 0.2, wobble: 0.5, wobbleRate: 4, tilt: 2.3, gain: 1 }
  })
}
