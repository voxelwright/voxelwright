export default function (villager, api) {
  const { THREE } = api
  const felt = (c) => new THREE.MeshStandardMaterial({ color: c, roughness: 1 })
  const bead = (c) => new THREE.MeshStandardMaterial({ color: c, roughness: 0.25 })
  const add = (parent, geo, mat, x, y, z) => { const m = new THREE.Mesh(geo, mat); m.position.set(x, y, z); parent.add(m); return m }
  const wool = felt('#efe6d2'), face = felt('#b9a38b'), hoof = felt('#6b5648'), thread = felt('#4a3a32')

  // Woolly body: long, wide cloud on sturdy legs
  add(villager, new THREE.SphereGeometry(0.17, 28, 18), wool, 0, 0.26, 0).scale.set(1, 0.9, 1.15)
  const puffs = [[0.12, 0.3, -0.05], [-0.12, 0.3, -0.05], [0.11, 0.2, -0.14], [-0.11, 0.2, -0.14], [0, 0.34, -0.1], [0, 0.2, -0.2], [0.13, 0.22, 0.06], [-0.13, 0.22, 0.06]]
  for (const [x, y, z] of puffs) add(villager, new THREE.SphereGeometry(0.075, 16, 12), wool, x, y, z)
  add(villager, new THREE.SphereGeometry(0.055, 14, 10), wool, 0, 0.24, -0.25) // tail puff
  // Neck: wool ruff plus a short face-coloured neck
  add(villager, new THREE.SphereGeometry(0.085, 18, 12), wool, 0, 0.32, 0.14)
  add(villager, new THREE.CapsuleGeometry(0.05, 0.07, 6, 10), face, 0, 0.35, 0.17)

  const head = villager.tag(new THREE.Group(), 'head')
  head.position.set(0, 0.4, 0.2)
  villager.add(head)
  add(head, new THREE.SphereGeometry(0.09, 24, 16), face, 0, 0, 0)
  const muzzle = add(head, new THREE.CapsuleGeometry(0.042, 0.05, 6, 12), face, 0, -0.03, 0.085)
  muzzle.rotation.x = Math.PI / 2
  add(head, new THREE.SphereGeometry(0.015, 12, 8), bead('#3a2a26'), 0, -0.012, 0.148)
  const smile = add(head, new THREE.TorusGeometry(0.02, 0.004, 6, 14, Math.PI), thread, 0, -0.045, 0.13)
  smile.rotation.z = Math.PI
  smile.rotation.x = -0.4
  for (const s of [-1, 1]) {
    const ear = add(head, new THREE.CapsuleGeometry(0.028, 0.07, 6, 10), face, s * 0.1, 0.02, -0.01)
    ear.rotation.z = -s * 1.87
    ear.scale.set(1, 1, 0.5)
    const eye = villager.tag(add(head, new THREE.CylinderGeometry(0.019, 0.019, 0.012, 16), bead('#1f1714'), s * 0.04, 0.03, 0.08), s < 0 ? 'eye-left' : 'eye-right')
    eye.rotation.x = Math.PI / 2 - 0.2
  }
  // Carer's mark: one bent stocking cap flopping to +x, pom-pom on its tip
  const capGeo = new THREE.ConeGeometry(0.078, 0.26, 20, 12)
  const p = capGeo.attributes.position
  for (let i = 0; i < p.count; i++) {
    const t = (p.getY(i) + 0.13) / 0.26
    p.setXYZ(i, p.getX(i) + 0.2 * t * t, p.getY(i) - 0.1 * t * t * t, p.getZ(i))
  }
  capGeo.computeVertexNormals()
  const capMat = felt('#9a8a7a')
  villager.mark(add(head, capGeo, capMat, 0, 0.15, -0.03))
  villager.mark(add(head, new THREE.SphereGeometry(0.042, 16, 12), capMat, 0.2, 0.18, -0.03))

  for (const s of [-1, 1]) {
    const hand = villager.tag(add(villager, new THREE.CapsuleGeometry(0.034, 0.05, 6, 10), face, s * 0.19, 0.22, 0.04), s < 0 ? 'hand-left' : 'hand-right')
    hand.rotation.z = s * 0.5
    const foot = villager.tag(new THREE.Group(), s < 0 ? 'foot-left' : 'foot-right')
    foot.position.set(s * 0.075, 0, 0.02)
    villager.add(foot)
    add(foot, new THREE.CylinderGeometry(0.03, 0.034, 0.11, 12), face, 0, 0.065, 0)
    add(foot, new THREE.SphereGeometry(0.04, 14, 10), hoof, 0, 0.02, 0.008).scale.set(1, 0.6, 1.25)
  }

  // Prop: a big hot-water bottle held at the left hand
  const bottle = villager.tag(new THREE.Group(), 'held')
  bottle.position.set(-0.2, 0.2, 0.09)
  villager.add(bottle)
  add(bottle, new api.RoundedBoxGeometry(0.14, 0.2, 0.05, 3, 0.022), felt('#d9a9a0'), 0, 0.03, 0)
  add(bottle, new THREE.CylinderGeometry(0.02, 0.024, 0.05, 12), felt('#8a7060'), 0, 0.15, 0)

  villager.gait({ speed: 0.3, stride: 0.2, bob: 0.03, stepped: true, fidget: 0.3 })
  // voice: slightly lower, slower and breathier than the start, for a soft, unhurried bedside manner.
  villager.about({
    name: 'Bramwell', role: 'carer', bark: 'Kettle is on and the blanket is warm. Sit down and tell me where it hurts.',
    voice: { pitch: 220, range: 2.8, formant: 1.1, rate: 8.5, breath: 0.3, wobble: 0.3, wobbleRate: 5, tilt: 2.1, gain: 0.86 },
  })
}
