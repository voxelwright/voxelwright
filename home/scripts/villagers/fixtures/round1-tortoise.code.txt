export default function (villager, api) {
  const { THREE } = api
  const felt = (c) => new THREE.MeshStandardMaterial({ color: c, roughness: 1 })
  const bead = (c) => new THREE.MeshStandardMaterial({ color: c, roughness: 0.25 })
  const add = (parent, geo, mat, x, y, z) => { const m = new THREE.Mesh(geo, mat); m.position.set(x, y, z); parent.add(m); return m }

  const skin = felt('#c9bb8f'), shellC = felt('#9c6b3a'), patchC = felt('#d9b878')
  const rimC = felt('#7a512c'), cream = felt('#f3e8d0'), plum = felt('#5a4560'), dark = felt('#4a3426')
  const blue = felt('#6f8fa8'), sage = felt('#9bb08a'), mustard = felt('#d6a444')
  const accent = felt('#c8553d')

  // soft belly / body under the shell
  add(villager, new THREE.SphereGeometry(0.2, 28, 16), skin, 0, 0.09, 0.0).scale.set(1, 0.5, 1.05)

  // domed shell
  const SY = 1.25, R = 0.23, CY = 0.125, CZ = -0.02
  const shell = add(villager, new THREE.SphereGeometry(R, 32, 20, 0, Math.PI * 2, 0, Math.PI / 2), shellC, 0, CY, CZ)
  shell.scale.set(1, SY, 1.05)
  const rim = add(villager, new THREE.TorusGeometry(R * 1.0, 0.022, 10, 36), rimC, 0, CY, CZ)
  rim.rotation.x = Math.PI / 2
  rim.scale.set(1, 1.05, 1)

  // hex patches on the shell
  const up = new THREE.Vector3(0, 1, 0)
  const patch = (el, az, size) => {
    const dx = Math.cos(el) * Math.sin(az), dy = Math.sin(el), dz = Math.cos(el) * Math.cos(az)
    const x = R * dx, y = CY + R * SY * dy, z = CZ + R * 1.05 * dz
    const n = new THREE.Vector3(x / (R * R), (y - CY) / (R * SY * R * SY), (z - CZ) / (R * 1.05 * R * 1.05)).normalize()
    const m = add(villager, new THREE.CylinderGeometry(size, size, 0.02, 6), patchC, x * 1.0, y, z)
    m.quaternion.setFromUnitVectors(up, n)
    return m
  }
  patch(Math.PI / 2, 0, 0.065)
  for (let i = 0; i < 5; i++) patch(0.62, (i / 5) * Math.PI * 2 + 0.3, 0.052)

  // tail
  const tail = add(villager, new THREE.ConeGeometry(0.03, 0.08, 12), skin, 0, 0.06, -0.255)
  tail.rotation.x = -Math.PI / 2

  // neck + scarf
  add(villager, new THREE.SphereGeometry(0.065, 18, 12), skin, 0, 0.14, 0.22).scale.set(1, 0.95, 1)
  const scarf = add(villager, new THREE.TorusGeometry(0.07, 0.022, 10, 28), accent, 0, 0.135, 0.22)
  scarf.rotation.x = Math.PI / 2
  add(villager, new THREE.CapsuleGeometry(0.02, 0.07, 6, 10), accent, 0.045, 0.09, 0.285).rotation.z = 0.12

  // head
  const head = villager.tag(new THREE.Group(), 'head')
  head.position.set(0, 0.22, 0.28)
  villager.add(head)
  add(head, new THREE.SphereGeometry(0.085, 28, 18), skin, 0, 0, 0).scale.set(1.05, 0.95, 1.05)
  add(head, new THREE.SphereGeometry(0.035, 16, 10), skin, 0, -0.025, 0.07).scale.set(1.3, 0.8, 0.8)
  // stitched smile
  const smile = add(head, new THREE.TorusGeometry(0.022, 0.004, 6, 14, Math.PI), dark, 0, -0.025, 0.093)
  smile.rotation.z = Math.PI
  // sleepy brows
  for (const s of [-1, 1]) {
    const brow = add(head, new THREE.CapsuleGeometry(0.006, 0.026, 4, 8), cream, s * 0.04, 0.05, 0.075)
    brow.rotation.z = Math.PI / 2 + s * -0.2
    const eye = villager.tag(add(head, new THREE.CylinderGeometry(0.014, 0.014, 0.01, 16), bead('#241a18'), s * 0.038, 0.016, 0.082), s < 0 ? 'eye-left' : 'eye-right')
    eye.rotation.x = Math.PI / 2
    const lens = add(head, new THREE.TorusGeometry(0.026, 0.004, 6, 20), dark, s * 0.038, 0.016, 0.092)
  }
  add(head, new THREE.CylinderGeometry(0.003, 0.003, 0.025, 6), dark, 0, 0.02, 0.094).rotation.z = Math.PI / 2
  // jazz beret
  const beret = add(head, new THREE.SphereGeometry(0.07, 20, 12), plum, 0.02, 0.075, -0.005)
  beret.scale.set(1.15, 0.42, 1.1)
  beret.rotation.z = -0.15
  add(head, new THREE.CylinderGeometry(0.006, 0.008, 0.02, 8), plum, 0.025, 0.103, -0.005)

  // hands and feet
  for (const s of [-1, 1]) {
    const hand = villager.tag(add(villager, new THREE.CapsuleGeometry(0.03, 0.04, 6, 10), skin, s * 0.21, 0.1, 0.07), s < 0 ? 'hand-left' : 'hand-right')
    hand.rotation.z = s * 0.6
    villager.tag(add(villager, new THREE.SphereGeometry(0.05, 16, 10), skin, s * 0.11, 0.03, 0.1), s < 0 ? 'foot-left' : 'foot-right').scale.set(1, 0.65, 1.3)
  }

  // held: little desk calendar
  const cal = villager.tag(new THREE.Group(), 'held')
  cal.position.set(-0.22, 0.14, 0.12)
  cal.rotation.y = 0.4
  villager.add(cal)
  add(cal, new api.RoundedBoxGeometry(0.11, 0.13, 0.016, 3, 0.006), cream, 0, 0.04, 0)
  add(cal, new api.RoundedBoxGeometry(0.11, 0.035, 0.02, 3, 0.006), blue, 0, 0.087, 0.002)
  const cols = [sage, mustard, sage, blue, mustard, sage]
  for (let i = 0; i < 6; i++) {
    add(cal, new THREE.BoxGeometry(0.022, 0.022, 0.006), cols[i], -0.032 + (i % 3) * 0.032, 0.05 - Math.floor(i / 3) * 0.032, 0.011)
  }

  villager.gait({ speed: 0.2, stride: 0.14, bob: 0.012, stepped: true, fidget: 0.25 })
  villager.about({
    name: 'Brindle',
    role: 'keeper',
    bark: "Dentist on Tuesday, Alex, and there's a quiet night free Friday. Shall I pencil in some jazz?",
    voice: { pitch: 100, range: 2.5, formant: 0.85, rate: 5, breath: 0.25, wobble: 0.5, wobbleRate: 4, tilt: 2.2, gain: 0.95 }
  })
}
