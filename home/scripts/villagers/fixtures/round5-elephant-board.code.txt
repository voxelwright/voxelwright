export default function (villager, api) {
  const { THREE } = api
  const felt = (c) => new THREE.MeshStandardMaterial({ color: c, roughness: 1 })
  const bead = (c) => new THREE.MeshStandardMaterial({ color: c, roughness: 0.25 })
  const add = (parent, geo, mat, x, y, z) => { const m = new THREE.Mesh(geo, mat); m.position.set(x, y, z); parent.add(m); return m }
  const grey = felt('#a7a199'), lightGrey = felt('#bdb7ae'), pinkIn = felt('#c9aea6'), cream = felt('#f1e6cf')
  const ink = bead('#241a18')

  // Body: a heavy barrel resting on the floor, head set forward of it.
  add(villager, new THREE.SphereGeometry(0.2, 32, 20), grey, 0, 0.2, -0.02).scale.set(1, 0.95, 1)
  add(villager, new THREE.SphereGeometry(0.1, 20, 12), lightGrey, 0, 0.17, 0.12).scale.set(1, 0.9, 0.4)
  // Short tail at the back.
  const tail = add(villager, new THREE.CapsuleGeometry(0.015, 0.07, 4, 8), grey, 0, 0.17, -0.22)
  tail.rotation.x = -0.4

  const head = villager.tag(new THREE.Group(), 'head')
  head.position.set(0, 0.4, 0.07)
  villager.add(head)
  add(head, new THREE.SphereGeometry(0.14, 32, 20), grey, 0, 0, 0).scale.set(1.05, 0.95, 1)

  // Big ear flaps.
  for (const s of [-1, 1]) {
    const ear = add(head, new THREE.SphereGeometry(0.13, 20, 14), grey, s * 0.165, -0.005, -0.02)
    ear.scale.set(0.3, 1, 0.85)
    ear.rotation.y = s * 0.35
    add(ear, new THREE.SphereGeometry(0.1, 16, 10), pinkIn, s * 0.2, 0, 0.1).scale.set(0.2, 1, 0.8)
  }

  // Trunk in two bent segments, reaching forward and down.
  const t1 = add(head, new THREE.CylinderGeometry(0.05, 0.043, 0.13, 16), grey, 0, -0.06, 0.16)
  t1.rotation.x = -0.75
  add(head, new THREE.SphereGeometry(0.043, 14, 10), grey, 0, -0.115, 0.205)
  const t2 = add(head, new THREE.CylinderGeometry(0.043, 0.034, 0.11, 16), grey, 0, -0.165, 0.225)
  t2.rotation.x = -0.35
  add(head, new THREE.SphereGeometry(0.034, 14, 10), grey, 0, -0.215, 0.24)
  // Tusks.
  for (const s of [-1, 1]) {
    const tusk = add(head, new THREE.ConeGeometry(0.02, 0.11, 12), cream, s * 0.075, -0.08, 0.17)
    tusk.rotation.x = 1.2
    tusk.rotation.z = s * -0.15
  }

  // Eyes.
  for (const s of [-1, 1]) {
    const eye = villager.tag(add(head, new THREE.SphereGeometry(0.02, 14, 10), ink, s * 0.06, 0.03, 0.125), s < 0 ? 'eye-left' : 'eye-right')
    eye.scale.set(1, 1, 0.8)
  }
  // Wise brow tufts in cream.
  for (const s of [-1, 1]) {
    const brow = add(head, new THREE.CapsuleGeometry(0.008, 0.04, 4, 8), cream, s * 0.06, 0.063, 0.12)
    brow.rotation.z = Math.PI / 2 + s * 0.2
  }

  // The tutor's mark: a mortarboard on a short cap, with a tassel.
  const board = felt('#3d3d44')
  villager.mark(add(head, new THREE.CylinderGeometry(0.085, 0.09, 0.09, 24), board, 0, 0.16, -0.04))
  villager.mark(add(head, new THREE.BoxGeometry(0.42, 0.02, 0.42), board, 0, 0.215, -0.04))
  villager.mark(add(head, new THREE.CylinderGeometry(0.01, 0.01, 0.09, 8), board, 0.19, 0.17, 0.15))
  villager.mark(add(head, new THREE.SphereGeometry(0.026, 12, 10), board, 0.19, 0.115, 0.15))

  // Hands and feet.
  for (const s of [-1, 1]) {
    const hand = villager.tag(add(villager, new THREE.CapsuleGeometry(0.036, 0.05, 6, 10), grey, s * 0.2, 0.19, 0.05), s < 0 ? 'hand-left' : 'hand-right')
    hand.rotation.z = s * 0.4
    villager.tag(add(villager, new THREE.CylinderGeometry(0.058, 0.064, 0.07, 18), grey, s * 0.1, 0.035, 0.03), s < 0 ? 'foot-left' : 'foot-right')
  }

  // Prop: a rolled history scroll with wooden ends.
  const scroll = villager.tag(new THREE.Group(), 'held')
  scroll.position.set(-0.235, 0.18, 0.06)
  villager.add(scroll)
  add(scroll, new THREE.CylinderGeometry(0.03, 0.03, 0.24, 16), cream, 0, 0, 0)
  for (const y of [-0.125, 0.125]) add(scroll, new THREE.CylinderGeometry(0.036, 0.036, 0.02, 14), felt('#a8703a'), 0, y, 0)

  villager.gait({ speed: 0.26, stride: 0.18, bob: 0.025, stepped: true, fidget: 0.2 })
  // voice: lower, slower and rounder than the start for a big, old, unhurried elephant; a touch more wobble for age.
  villager.about({
    name: 'Ambrose', role: 'tutor', bark: 'Sit down, you two. The story of how this town got its name is a good one, and it is shorter than you fear.',
    voice: { pitch: 135, range: 2.5, formant: 0.85, rate: 6.5, breath: 0.2, wobble: 0.7, wobbleRate: 4.5, tilt: 2.2, gain: 1.0 },
  })
}
